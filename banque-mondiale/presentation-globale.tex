\section{Histoire et mission}
\begin{frame}{Histoire et mission (1/2)}
\begin{itemize}
\item Création de la BIRD à Bretton Woods en 1944, opérations dès 1946, premier prêt à la France en 1947
\item Passage de la reconstruction de l'Europe au développement dans les années 1950 et 1960
\item Années 1980 : ajustement structurel et crise de la dette
\item 2013 : deux objectifs jumeaux, fin de l'extrême pauvreté et prospérité partagée
\end{itemize}
\end{frame}

\begin{frame}{Histoire et mission (2/2)}
\begin{itemize}
\item Depuis 2023 : vision d'un monde sans pauvreté sur une planète vivable
\item Mission : financer le développement par prêts, dons, garanties, conseil et connaissances
\end{itemize}
\end{frame}

\section{Groupe et institutions}
\begin{frame}{Groupe et institutions (1/2)}
\begin{itemize}
\item Cinq institutions : BIRD, IDA, SFI, AMGI, CIRDI
\item BIRD : prêts aux pays à revenu intermédiaire
\item IDA : prêts très concessionnels et dons aux pays les plus pauvres
\item SFI : secteur privé. AMGI : garanties contre le risque politique
\end{itemize}
\end{frame}

\begin{frame}{Groupe et institutions (2/2)}
\begin{itemize}
\item CIRDI : arbitrage entre investisseurs et États
\item La Banque mondiale au sens strict : BIRD et IDA
\end{itemize}

\section{Gouvernance}
\begin{frame}{Gouvernance (1/2)}
\begin{itemize}
\item Conseil des gouverneurs : un gouverneur par pays membre, réunion annuelle
\item Conseil d'administration : 25 administrateurs, approbation des prêts et politiques
\item Droits de vote fondés sur les parts de capital, poids des grands actionnaires
\item Présidence choisie par le conseil, traditionnellement un ressortissant américain
\end{itemize}
\end{frame}
