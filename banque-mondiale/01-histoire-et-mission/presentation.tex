\begin{frame}{Histoire et mission (1/2)}
\begin{itemize}
\item Création de la BIRD à Bretton Woods en 1944, opérations dès 1946, premier prêt à la France en 1947
\item Passage de la reconstruction de l'Europe au développement dans les années 1950 et 1960
\item Années 1980 : ajustement structurel et crise de la dette
\item 2013 : deux objectifs jumeaux, fin de l'extrême pauvreté et prospérité partagée
\end{itemize}
\end{frame}

\end{itemize}
