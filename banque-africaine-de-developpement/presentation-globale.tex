\section{Histoire et mission}
\begin{frame}{Histoire et mission (1/2)}
\begin{itemize}
\item Créée en 1964 (accord de Khartoum), opérations en 1966, siège à Abidjan
\item Siège temporairement à Tunis de 2003 à 2014 pour raisons de sécurité
\item Mission : réduire la pauvreté et favoriser la croissance durable en Afrique
\item Membres : pays africains et pays non régionaux (environ 81, à vérifier)
\end{itemize}
\end{frame}

\begin{frame}{Histoire et mission (2/2)}
\begin{itemize}
\item Notation AAA des principales agences
\end{itemize}

\section{Groupe et institutions}
\begin{frame}{Groupe et institutions}
\begin{itemize}
\item Trois guichets : Banque africaine de développement, Fonds africain de développement, Fonds spécial du Nigeria
\item BAD : prêts aux pays à revenu intermédiaire et au secteur privé
\item FAD : prêts concessionnels et dons pour les pays les plus pauvres, reconstitué périodiquement
\item Fonds spécial du Nigeria : financements de projets à conditions favorables
\end{itemize}
\end{frame}

\section{Gouvernance}
\begin{frame}{Gouvernance}
\begin{itemize}
\item Conseil des gouverneurs : un gouverneur par pays membre, assemblées annuelles
\item Conseil d'administration de 20 administrateurs, dont 13 pour les pays régionaux (à vérifier)
\item Pays régionaux : majorité du capital et des voix, environ 60 \% (à vérifier)
\item Président élu par les gouverneurs pour cinq ans : Akinwumi Adesina de 2015 à 2025, puis Sidi Ould Tah (à vérifier)
\end{itemize}
\end{frame}
